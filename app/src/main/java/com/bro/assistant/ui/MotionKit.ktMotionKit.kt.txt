        SecretBox.saveSetting(context, "anim_$slot", id.toString())
    }

/**
 * Modifier that applies any of the 104 animations dynamically with Reduced-Motion support.
 */
@Composable
fun Modifier.applyBroMotion(
    animId: Int,
    reducedMotion: Boolean = false
): Modifier {
    if (reducedMotion) return this

    val progress = remember { Animatable(0f) }
    LaunchedEffect(animId) {
        progress.snapTo(0f)
        progress.animateTo(
            targetValue = 1f,
            animationSpec = spring(
                dampingRatio = Spring.DampingRatioMediumBouncy,
                stiffness = Spring.StiffnessLow
            )
        )
    }

    val infinite = rememberInfiniteTransition(label = "bro_loop_$animId")
    val loopPhase by infinite.animateFloat(
        initialValue = 0f,
        targetValue = 1f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = 1600 + ((animId % 7) * 250), easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "loop_phase"
    )

    val p = progress.value
    val wave = sin(loopPhase * 2f * PI.toFloat())

    return this.graphicsLayer {
        when {
            animId in 1..46 -> {
                // Entrance variations parameterized by ID
                alpha = p.coerceIn(0f, 1f)
                val dir = if (animId % 2 == 0) 1f else -1f
                when (animId % 5) {
                    0 -> translationY = (1f - p) * 60f * dir
                    1 -> translationX = (1f - p) * 70f * dir
                    2 -> {
                        scaleX = 0.65f + (0.35f * p)
                        scaleY = 0.65f + (0.35f * p)
                    }
                    3 -> rotationZ = (1f - p) * 18f * dir
                    else -> {
                        translationY = (1f - p) * 40f
                        scaleX = 0.85f + (0.15f * p)
                        scaleY = 0.85f + (0.15f * p)
                    }
                }
            }
            animId in 47..74 -> {
                // Emphasis pulse
                val emp = sin(p * PI.toFloat())
                scaleX = 1f + (emp * 0.12f)
                scaleY = 1f + (emp * 0.12f)
                rotationZ = if (animId % 2 == 0) emp * 6f else -emp * 6f
            }
            else -> {
                // Continuous 30 Loops
                when (animId % 4) {
                    0 -> translationY = wave * 6f
                    1 -> {
                        val s = 1f + (wave * 0.05f)
                        scaleX = s
                        scaleY = s
                    }
                    2 -> rotationZ = wave * 8f
                    else -> alpha = 0.82f + (wave * 0.18f)
                }
            }
        }
    }
}

@Composable
fun MotionLabPanel(
    context: Context,
    reducedMotion: Boolean,
    onReducedMotionChanged: (Boolean) -> Unit
) {
    var selectedSlot by remember { mutableStateOf("messages") }
    var selectedCategory by remember { mutableStateOf(MotionCategory.ENTRANCE) }
    var activeAnimId by remember(selectedSlot) {
        mutableIntStateOf(
            when (selectedSlot) {
                "messages" -> MotionKit.getSlotAnim(context, "messages", 1).id
                "cards" -> MotionKit.getSlotAnim(context, "cards", 5).id
                "send" -> MotionKit.getSlotAnim(context, "send", 47).id
                else -> MotionKit.getSlotAnim(context, "logo", 75).id
            }
        )
    }

    val slots = listOf(
        "messages" to "MESSAGES",
        "cards" to "HUD CARDS",
        "send" to "SEND BTN",
        "logo" to "RADAR LOGO"
    )

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(12.dp),
        verticalArrangement = Arrangement.spacedBy(10.dp)
    ) {
        // Live Preview Stage
        TacticalCard(
            modifier = Modifier
                .fillMaxWidth()
                .applyBroMotion(activeAnimId, reducedMotion),
            accentColor = BroDesign.VoltLime
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Column {
                    Text("104-ANIMATION MOTION LAB", style = BroDesign.HeaderStyle.copy(fontSize = 15.sp))
                    Text(
                        "SLOT: ${selectedSlot.uppercase()} -> #${activeAnimId} ${MotionKit.findById(activeAnimId).name}",
                        style = BroDesign.HudLabelStyle.copy(color = BroDesign.CyberCyan)
                    )
                }
                Box(
                    modifier = Modifier
                        .clip(BroDesign.PillShape)
                        .background(if (reducedMotion) BroDesign.HyperOrange else BroDesign.ElevatedTactile)
                        .border(1.dp, BroDesign.BorderGrid, BroDesign.PillShape)
                        .clickable { onReducedMotionChanged(!reducedMotion) }
                        .padding(horizontal = 8.dp, vertical = 5.dp)
                ) {
                    Text(
                        text = if (reducedMotion) "REDUCED: ON" else "MOTION: FULL",
                        style = BroDesign.HudLabelStyle.copy(
                            color = if (reducedMotion) Color.Black else BroDesign.VoltLime,
                            fontSize = 9.sp
                        )
                    )
                }
            }
        }

        // Target Slot Selector (Messages, Cards, Send Button, Logo)
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            slots.forEach { (key, label) ->
                val sel = selectedSlot == key
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .clip(BroDesign.PillShape)
                        .background(if (sel) BroDesign.VoltLime else BroDesign.CarbonSurface)
                        .border(1.dp, if (sel) BroDesign.VoltLime else BroDesign.BorderGrid, BroDesign.PillShape)
                        .clickable { selectedSlot = key }
                        .padding(vertical = 8.dp),
                    contentAlignment = Alignment.Center
                ) {
                    Text(
                        text = label,
                        style = BroDesign.HudLabelStyle.copy(
                            color = if (sel) Color.Black else BroDesign.CrispWhite,
                            fontSize = 10.sp
                        )
                    )
                }
            }
        }

        // Category Selector (46 Entrances / 28 Emphasis / 30 Loops)
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            MotionCategory.values().forEach { cat ->
                val sel = selectedCategory == cat
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .clip(BroDesign.PillShape)
                        .background(if (sel) BroDesign.CyberCyan else BroDesign.ElevatedTactile)
                        .clickable { selectedCategory = cat }
                        .padding(vertical = 6.dp),
                    contentAlignment = Alignment.Center
                ) {
                    Text(
                        text = cat.label,
                        style = BroDesign.HudLabelStyle.copy(
                            color = if (sel) Color.Black else BroDesign.CrispWhite,
                            fontSize = 10.sp
                        )
                    )
                }
            }
        }

        // List of Animations in Selected Category
        val filtered = remember(selectedCategory) {
            MotionKit.all104.filter { it.category == selectedCategory }
        }

        LazyColumn(
            modifier = Modifier.weight(1f),
            verticalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            items(filtered, key = { it.id }) { spec ->
                val isCurrent = spec.id == activeAnimId
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .clip(BroDesign.CardShape)
                        .background(if (isCurrent) BroDesign.ElevatedTactile else BroDesign.CarbonSurface)
                        .border(
                            1.dp,
                            if (isCurrent) BroDesign.VoltLime else BroDesign.BorderGrid,
                            BroDesign.CardShape
                        )
                        .clickable {
                            activeAnimId = spec.id
                            MotionKit.setSlotAnim(context, selectedSlot, spec.id)
                        }
                        .padding(12.dp),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = "#${spec.id}  ${spec.name}",
                        style = BroDesign.BodyStyle.copy(
                            color = if (isCurrent) BroDesign.VoltLime else BroDesign.CrispWhite,
                            fontSize = 14.sp
                        )
                    )
                    Text(
                        text = if (isCurrent) "ACTIVE" else "APPLY",
                        style = BroDesign.HudLabelStyle.copy(
                            color = if (isCurrent) BroDesign.VoltLime else BroDesign.MutedSteel,
                            fontSize = 10.sp
                        )
                    )
                }
            }
        }
    }
}
